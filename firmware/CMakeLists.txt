cmake_minimum_required(VERSION 3.21)
set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../third_party/pico-sdk" CACHE PATH "Pico SDK")
set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
set(PICO_BOARD waveshare_rp2350b_plus_w CACHE STRING "Board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Platform")
set(PICO_CXX_ENABLE_EXCEPTIONS 0)
set(PICO_CXX_ENABLE_RTTI 0)
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ai_debugprobe VERSION 0.3.0 LANGUAGES C CXX ASM)
set(PROBE_SYS_CLOCK_MHZ 300 CACHE STRING "Probe system clock: 150 or 300 MHz")
set_property(CACHE PROBE_SYS_CLOCK_MHZ PROPERTY STRINGS 150 300)
if(NOT PROBE_SYS_CLOCK_MHZ MATCHES "^(150|300)$")
    message(FATAL_ERROR "PROBE_SYS_CLOCK_MHZ must be 150 or 300")
endif()
math(EXPR PROBE_CPU_CLOCK_HZ "${PROBE_SYS_CLOCK_MHZ} * 1000000")
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
pico_sdk_init()
set(DAP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../third_party/CMSIS-DAP/Firmware")
add_executable(ai_debugprobe main.cpp usb_descriptors.cpp packet_guard.cpp logic_analyzer.cpp
    ${DAP_ROOT}/Source/DAP.c ${DAP_ROOT}/Source/SW_DP.c)
target_include_directories(ai_debugprobe PRIVATE ${CMAKE_CURRENT_LIST_DIR}
    ${DAP_ROOT}/Include
    ${PICO_SDK_PATH}/src/rp2_common/cmsis/stub/CMSIS/Core/Include)
target_link_libraries(ai_debugprobe PRIVATE pico_stdlib pico_unique_id tinyusb_device hardware_watchdog pico_multicore hardware_pio hardware_dma hardware_clocks hardware_vreg)
target_compile_definitions(ai_debugprobe PRIVATE PROBE_CPU_CLOCK_HZ=${PROBE_CPU_CLOCK_HZ}U PROBE_FIRMWARE_VERSION="${PROJECT_VERSION}")
target_compile_options(ai_debugprobe PRIVATE -Wall -Wextra -Wno-unused-parameter)
pico_enable_stdio_usb(ai_debugprobe 0)
pico_enable_stdio_uart(ai_debugprobe 0)
pico_set_program_name(ai_debugprobe "AI CMSIS-DAP SWD Probe")
pico_set_program_version(ai_debugprobe "${PROJECT_VERSION}")
pico_add_extra_outputs(ai_debugprobe)
